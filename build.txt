# Generate container
sudo podman build --cap-add=all --security-opt=label=type:container_runtime_t --device /dev/fuse -t ghcr.io/trexx/fedora-minimal-bootc -f dockerfile .

# Optimize container
sudo podman run --rm --privileged -v /var/lib/containers:/var/lib/containers quay.io/fedora/fedora-bootc:44 /usr/libexec/bootc-base-imagectl rechunk ghcr.io/trexx/fedora-minimal-bootc:latest ghcr.io/trexx/fedora-minimal-bootc:chunked

# Build Anaconda ISO
sudo podman run     --rm     -it     --privileged     --pull=newer     --security-opt label=type:unconfined_t -v ./config.toml:/config.toml:ro -v ./output:/output     -v /var/lib/containers/storage:/var/lib/containers/storage     quay.io/centos-bootc/bootc-image-builder:latest --type anaconda-iso --target-arch amd64 --rootfs xfs ghcr.io/trexx/fedora-minimal-bootc:chunked
